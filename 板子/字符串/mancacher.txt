#include<bits/stdc++.h>
using namespace std;
using ll=long long;
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	int T;
	cin>>T;
	while(T--){
		string s="";
		cin>>s;
		ll ans=0;
		string s1="$";
		for(char c:s){
			s1+="#";
			s1+=c;
		}
		s1+="#$";
		int n=s1.size(),ct=0,r=0;
		vector<int> p(n,0);
		for(int i=1;i<=n-2;i++){
			int mir=ct*2-i;
			if(i<r)p[i]=min(r-i,p[mir]);
			while(s1[i+p[i]+1]==s1[i-p[i]-1])p[i]++;
			if(i+p[i]>r){
				ct=i;
				r=i+p[i];
			}
			ans+=(p[i]+1)/2;
		}
		cout<<ans<<endl;
	}
	return 0;
}
