vector<int> d1(n),d2(n);
	for(int i=0,l=0,r=-1;i<n;i++){
		int k;
		if(i>r)k=1;
		else k=min(d1[r+l-i],1+r-i);
		while(i-k>=0&&i+k<n&&s[i-k]==s[i+k])k++;
		d1[i]=k;
		if(i+k-1>r){
			r=i+k-1;
			l=i-k+1;
		}
	}
	for(int i=0,l=0,r=-1;i<n;i++){
		int k;
		if(i>r)k=0;
		else k=min(d2[r+l-i+1],1+r-i);
		while(i-k-1>=0&&i+k<n&&s[i-k-1]==s[i+k])k++;
		d2[i]=k;
		if(i+k-1>r){
			r=i+k-1;
			l=i-k;
		}
	}
