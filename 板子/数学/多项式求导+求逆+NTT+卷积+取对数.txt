#include<bits/stdc++.h>
using namespace std;
using ll=long long;
const int mod=998244353,root=3,maxn=1e5+5;
ll f[maxn],inf[maxn];
ll fastpow(ll a,ll b){
	ll ans=1;
	while(b){
		if(b&1)ans=ans*a%mod;
		a=a*a%mod;
		b=b>>1;
	}
	return ans;
}
ll mod_inv(ll a){
	return fastpow(a,mod-2);
}
void ntt(vector<ll> &a,bool invert){
	int n=a.size();
	for(int i=1,j=0;i<n;i++){
		int bit=n>>1;
		for(;j&bit;bit=bit>>1)j^=bit;
		j^=bit;
		if(i<j)swap(a[i],a[j]);
	}
	for(int len=2;len<=n;len=len<<1){
		ll wlen=fastpow(root,(mod-1)/len);
		if(invert)wlen=mod_inv(wlen);
		for(int i=0;i<n;i+=len){
			ll w=1;
			for(int j=0;j<len/2;j++){
				ll u=a[i+j];
				ll v=a[i+j+len/2]*w%mod;
				a[i+j]=(u+v)%mod;
				a[i+j+len/2]=(u-v+mod)%mod;
				w=w*wlen%mod;
			}
		}
	}
	if(invert){
		ll inn=mod_inv(n);
		for(ll &x:a)x=x*inn%mod;
	}
}
vector<ll> convolution(vector<ll> a,vector<ll> b){
	int len=a.size()+b.size()-1;
	int n=1;
	while(n<len)n=n<<1;
	a.resize(n);
	b.resize(n);
	ntt(a,false);
	ntt(b,false);
	for(int i=0;i<n;i++)a[i]=a[i]*b[i]%mod;
	ntt(a,true);
	a.resize(len);
	return a;
}
vector<ll> derivative(vector<ll> a){
	if(a.empty())return {};
	int n=a.size();
	vector<ll> res(n-1);
	for(int i=1;i<n;i++)res[i-1]=a[i]*i%mod;
	return res;
}
vector<ll> inverse(vector<ll> a,int m){
	assert(!a.empty()&&a[0]!= 0);
	vector<ll> inv={mod_inv(a[0])};
	while(inv.size()<m){
		int nextlen=min(2*(int)inv.size(),m);
		vector<ll> a_cut(a.begin(),a.begin()+min((int)a.size(),nextlen));
		vector<ll> insq=convolution(inv,inv);
		insq.resize(nextlen);
		vector<ll> a_inv=convolution(a_cut,insq);
		a_inv.resize(nextlen);
		inv.resize(nextlen);
		for(int i=0;i<nextlen;i++){
			if(i<inv.size())inv[i]=(2*inv[i]-a_inv[i]+mod)%mod;
			else inv[i]=(mod-a_inv[i])%mod;
		}
	}
	return inv;
}
vector<ll> ln(vector<ll> a,int deg){
	assert(!a.empty()&&a[0]==1);
	vector<ll> da=derivative(a);
	vector<ll> a_inv=inverse(a,deg);
	vector<ll> pd=convolution(da,a_inv);
	if(pd.size()>deg)pd.resize(deg);
	vector<ll> res(deg+1,0);
	for(int i=1;i<=deg;i++){
		if(i-1<pd.size())res[i]=pd[i-1]*mod_inv(i)%mod;
	}
	return res;
}
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	int n;
	cin>>n;
	f[0]=1;
	for(int i=1;i<=n;i++)f[i]=f[i-1]*i%mod;
	inf[n]=mod_inv(f[n]);
	for(int i=n-1;i>=0;i--)inf[i]=inf[i+1]*(i+1)%mod;
	vector<ll> a(n+1);
	for(int i=0;i<=n;i++){
		ll exp=(ll)i*(i-1)/2;
		a[i]=fastpow(2,exp);
	}
	vector<ll> A(n+1);
	for(int i=0;i<=n;i++)A[i]=a[i]*inf[i]%mod;
	vector<ll> B=ln(A,n);
	ll bn=B[n]*f[n]%mod,an=a[n];
	ll ans=bn*mod_inv(an)%mod;
	cout<<ans<<endl;
	return 0;
}
