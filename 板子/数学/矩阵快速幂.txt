#include<bits/stdc++.h>
using namespace std;
typedef vector<vector<double>> mat;
mat mul1(const mat& A,const mat& B){
	mat C(9,vector<double>(9,0.0));
	for(int i=0;i<9;i++){
		for (int k=0;k<9;k++){
			if(A[i][k]==0)continue;
			for(int j=0;j<9;j++){
				C[i][j]+=A[i][k]*B[k][j];
			}
		}
	}
	return C;
}
vector<double> mul2(const mat& A,const vector<double>& v){
	vector<double> res(9,0.0);
	for(int i=0;i<9;i++){
		for(int j=0;j<9;j++){ 
			res[i]+=A[i][j]*v[j];
		}
	} 
	return res;
}
mat matpower(mat base,long long exp){
	mat res(9,vector<double>(9,0.0));
	for(int i=0;i<9;i++)res[i][i]=1.0;
	while(exp>0){
		if(exp&1)res=mul1(res,base);
		base=mul1(base,base);
		exp>>=1;
	}
	return res;
}
