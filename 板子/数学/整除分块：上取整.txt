#include<bits/stdc++.h>
using namespace std;
using ll=long long;
int ceill(int a,int b){
	return (a+b-1)/b;
}
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
    int T;
	cin>>T;
	while(T--){
		int n,m;
		cin>>n>>m;
		if(m%n==0){
			cout<<0<<endl;
		}else{
			int ans=0;
			if(m<n)ans=n-m;
			else{
				ans=1e9;
				for(int l=1,r;l<=n;l=r+1){
					r=(m-1)/((m-1)/l);
					ans=min(ans,ceill(m,l)*l-m+n-l);
				}
			}
			cout<<ans<<endl;
		}
	}
	return 0;
}
