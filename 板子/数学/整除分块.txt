#include<bits/stdc++.h>
using namespace std;
using ll=long long;
const int mod=998244353,inv=(mod+1)/2;
ll cal(ll n){
	ll res=0;
	ll l=1,r;
	while(l<=n){
		r=n/(n/l);
		res=(res+1LL*(r-l+1)%mod*(n/l)%mod)%mod;
		l=r+1;
	}
	return res;
}
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	ll n;
	cin>>n;
	ll a=cal(n);
	n=n%mod;
	a=(a-n+mod)%mod;
	ll b=((n-1)*(n-2)%mod)*inv%mod;
	ll c=(n-1)%mod;
	cout<<((b-a+mod)%mod+c)%mod;
	return 0;
}
