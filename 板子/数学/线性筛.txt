#include<bits/stdc++.h>
using namespace std;
vector<int> euler_sieve(int n){
	vector<int> p;
	vector<bool> not_p(n+1,false);
	for(int i=2;i<=n;i++){
		if(!not_p[i])p.push_back(i);
		for(int pp:p){
			if(i*pp>n)break;
		    not_p[i*pp]=true;
			if(i%pp==0)break;
		}
	}
	return p;
}
int main(){
	int n;
	cin>>n;
	vector<int> ans=euler_sieve(n);
	for(int num:ans)cout<<num<<" ";
	return 0;
}
