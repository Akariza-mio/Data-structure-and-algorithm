#include<bits/stdc++.h>
using namespace std;
using ll=long long;
const int maxn=1e6+5;
ll f[maxn],inf[maxn],ans[maxn],pre[maxn];
ll fastpow(ll a,ll b,ll mod){
	ll ans=1;
	while(b){
		if(b&1)ans=ans*a%mod;
		a=a*a%mod;
		b=b>>1;
	}
	return ans;
}
ll C(int n,int k,int mod){
	if(k<0||k>n)return 0;
	return f[n]*inf[k]%mod*inf[n-k]%mod;
}
void init(int n){
	f[0]=1;
	for(int i=1;i<=n;i++)f[i]=f[i-1]*i%mod;
	inf[n]=fastpow(f[n],mod-2);
	for(int i=n-1;i>=0;i--)inf[i]=inf[i+1]*(i+1)%mod;
}
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	int n,T,p;
	cin>>n>>T>>p;
	f[0]=inf[0]=1;
	for(int i=1;i<=n;i++){
		f[i]=f[i-1]*i%p;
		inf[i]=fastpow(f[i],p-2,p);
	}
	for(int i=0;i<=n;i++){
	    if(i>=2){
			ll t=fastpow(i,i-2,p);
			ll c=C(n,i,p);
			ans[i]=t*c%p;
		}
		else ans[i]=0;
		pre[i]=(i==0?ans[i]:(pre[i-1]+ans[i])%p);
	}
	while(T--){
		int l=0,r=0;
		cin>>l>>r;
		ll res=0;
		res=pre[r];
		if(l>0)res=(res-pre[l-1]+p)%p;
		cout<<res<<endl;
	}
	return 0;
}
