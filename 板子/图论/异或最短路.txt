#include<bits/stdc++.h>
using namespace std;
const int MAXN=1005,MAXW=1<<10;
vector<pair<int,int>>g[MAXN];
bool vis[MAXN][MAXW];
int basis[11],min_xor;
int minimize(int x){
    for(int i=10;i>=0;i--)
        if((x^basis[i])<x)x^=basis[i];
    return x;
}
void insert(int x){
    for(int i=10;i>=0;i--){
        if((x>>i)&1){
            if(!basis[i]){
                basis[i]=x;
                min_xor=min(min_xor,minimize(min_xor));
                break;
            }
            x^=basis[i];
        }
    }
}
void bfs(int s){
    queue<pair<int,int>>q;
    q.push({s,0});
    vis[s][0]=true;
    while(!q.empty()){
        auto[u,x]=q.front();q.pop();
        if(u==s&&x!=0)insert(x);
        for(auto[v,w]:g[u]){
            int nx=x^w;
            if(!vis[v][nx]){
                vis[v][nx]=true;
                q.push({v,nx});
                if(v==s)insert(nx);
            }
        }
    }
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    int n,m;cin>>n>>m;
    while(m--){
        int u,v,w;cin>>u>>v>>w;
        g[u].emplace_back(v,w);
    }
    memset(basis,0,sizeof(basis));
    memset(vis,0,sizeof(vis));
    min_xor=MAXW;
    bfs(1);
    for(int x=0;x<MAXW;x++)
        if(vis[n][x])min_xor=min(min_xor,minimize(x));
    cout<<(min_xor==MAXW?-1:min_xor)<<'\n';
}
