#include<iostream>
#include<vector>
#include<cstring>
using namespace std;
const int maxn=100005;
vector<int> g[maxn];
int match[maxn];
bool vis[maxn];
bool dfs(int u){
    for(int v:g[u]){
        if(!vis[v]){
            vis[v]=1;
            if(match[v]==-1||dfs(match[v])){
                match[v]=u;
                return 1;
            }
        }
    }
    return 0;
}
int main(){
    int T;
    cin>>T;
    while(T--){
        ios::sync_with_stdio(false);
        cin.tie(0);
        int n,m;
        cin>>n>>m;
        for(int i=1;i<=m;i++)g[i].clear();
        memset(match,-1,sizeof(match));
        for (int i=1;i<=m;i++) {
            int c=0;
            cin>>c;
            for(int j=1;j<=c;j++) {
                int dui=0;
                cin>>dui;
                g[i].push_back(dui);
            }
        }
        int ans=0;
        for(int i=1;i<=m;i++) {
            memset(vis,0,sizeof(vis));
            if(dfs(i))ans++;
        }
        cout<<ans<<endl;
    }
    return 0;
}
