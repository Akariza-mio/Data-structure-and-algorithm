#include<iostream>
#include<queue>
#include<vector>
const long long INF=1e18;
using namespace std;
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	long long n,m,s;
    cin>>n>>m>>s;
	vector<vector<pair<long long,long long>>> graph(n+1);
	vector<long long> dis(n+1,INF);
	dis[s]=0;
	for(long long i=1;i<=m;i++){
		long long u=0,v=0,w=0;
		cin>>u>>v>>w;
		graph[u].emplace_back(v,w);
	}
	priority_queue<pair<long long,long long>,vector<pair<long long,long long>>,greater<pair<long long,long long>>> pq;
	pq.emplace(0,s);
	while(!pq.empty()){
		auto [tp,u]=pq.top();
		pq.pop();
		if(tp>dis[u])continue;
		for(auto [v,w]:graph[u]){
			if(dis[u]+w<dis[v]){
				dis[v]=dis[u]+w;
				pq.emplace(dis[v],v);
			}
		}
	}
	for(long long i=1;i<=n;i++)cout<<dis[i]<<" ";
	return 0;
}
