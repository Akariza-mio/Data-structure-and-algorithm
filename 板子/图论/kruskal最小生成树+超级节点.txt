#include<iostream>
#include<algorithm>
#include<vector>
using namespace std;
struct E{
	int u,v,len;
};
const int maxn=10005;
int f[maxn],r[maxn],a[maxn];
void initiate(int n){
	for(int i=1;i<=n+1;i++){
		f[i]=i;
		r[i]=0;
	}
}
int find(int x){
	if(f[x]!=x)f[x]=find(f[x]);
	return f[x];
}
bool uni(int x,int y){
	int fx=find(x),fy=find(y);
	if(fx==fy)return 0;
	if(r[fx]<r[fy])f[fx]=fy;
	else{
		f[fy]=fx;
		if(r[fx]==r[fy])r[fx]++;
	}
	return 1;
}
bool cmp(E a,E b){
	return a.len<b.len;
}
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	int T;
	cin>>T;
	while(T--){
		int n=0;
		cin>>n;
		for(int i=1;i<=n;i++)cin>>a[i];
		vector<E> edges;
		for(int i=1;i<=n;i++){
			for(int j=1;j<=n;j++){
				int tp=0;
				cin>>tp;
				if(i<j)edges.push_back({i,j,tp});
			}
		}
		for(int i=1;i<=n;i++)edges.push_back({i,n+1,a[i]});
		sort(edges.begin(),edges.end(),cmp);
		initiate(n);
		int ans=0;
		for(int i=0;i<edges.size();i++){
			int u=edges[i].u,v=edges[i].v,len=edges[i].len;
            if(uni(u,v))ans+=len;
		}
		cout<<ans<<endl;
	}
	return 0;
}
