#include<bits/stdc++.h>
using namespace std;
const long long maxn=1000005,inf=1e10;
using ll=long long;
vector<pair<int,int>> g[maxn];
priority_queue<pair<int,int>,vector<pair<int,int>>,greater<pair<int,int>>> pq;
vector<ll> dis(maxn,inf);
void add(int u,int v,int w){
	g[u].push_back({v,w});
	g[v].push_back({u,w});
}
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	int n,m;
	cin>>n>>m;
	for(int i=1;i<=n;i++){
		for(int j=1;j<m;j++){
			int tp=0;
			cin>>tp;
			int up=(i==1)?0:(2*(i-2)*(m-1)+2*j-1);
			int down=(i==n)?(2*(n-1)*(m-1)+1):(2*(i-1)*(m-1)+2*j);
			add(up,down,tp);
		}
	}
	for(int i=1;i<n;i++){
		for(int j=1;j<=m;j++){
			int tp=0;
			cin>>tp;
			int l=(j==1)?(2*(n-1)*(m-1)+1):(2*(i-1)*(m-1)+2*(j-1));
			int r=(j==m)?0:(2*(i-1)*(m-1)+2*j-1);
			add(l,r,tp);
		}
	}
	for(int i=1;i<n;i++){
		for(int j=1;j<m;j++){
			int tp=0;
			cin>>tp;
			int a=2*(i-1)*(m-1)+2*j-1;
			int b=a+1;
			add(a,b,tp);
		}
	}
	dis[0]=0;
	pq.push({0,0});
	while(!pq.empty()){
		int d=pq.top().first,tp=pq.top().second;
		pq.pop();
		if(d>dis[tp])continue;
		for(auto [v,w]:g[tp]){
			if(dis[v]>dis[tp]+w){
				dis[v]=dis[tp]+w;
				pq.push({dis[v],v});
			}
		}
	}
	cout<<dis[2*(n-1)*(m-1)+1];
	return 0;
}
