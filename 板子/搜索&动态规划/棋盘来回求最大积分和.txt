#include<bits/stdc++.h>
using namespace std;
const int maxn=705;
long long point[maxn][maxn],dp[maxn][maxn];
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	int n,m;
	cin>>n>>m;
	for(int i=0;i<m;i++){
		int x=0,y=0,z=0;
		cin>>x>>y>>z;
		point[x][y]=z;
	}
	for(int step=2;step<=2*n;step++){
		for(int x1=min(n,step-1);x1>=max(1,step-n);x1--){
			for(int x2=min(n,step-1);x2>=max(1,step-n);x2--){
				int y1=step-x1,y2=step-x2;
				if(y1>n||y2>n)continue;
				long long tp=point[x1][y1];
				if(x1!=x2&&y1!=y2)tp+=point[x2][y2];
				dp[x1][x2]=max({dp[x1-1][x2],dp[x1][x2-1],dp[x1-1][x2-1],dp[x1][x2]})+tp;
			}
		}
	}
	cout<<dp[n][n]<<endl;
	return 0;
}
