#include<bits/stdc++.h>
using namespace std;
const int mod=1e9+7;
string k;
int memo[10005][100][2];
int dfs(int pos,int m,bool ti,int dd){
	if(pos==k.size())return m%dd==0?1:0;
	if(memo[pos][m][ti]!=-1)return memo[pos][m][ti];
	int ans=0;
	int lim=ti?(k[pos]-'0'):9;
	for(int i=0;i<=lim;i++){
		bool nextti=ti&&(lim==i);
		ans=(ans+dfs(pos+1,(m+i)%dd,nextti,dd))%mod;
	}
	memo[pos][m][ti]=ans;
	return ans;
}
int main(){
	int d;
	cin>>k>>d;
	memset(memo,-1,sizeof(memo));
	int ans=dfs(0,0,1,d);
	ans=(ans-1+mod)%mod;
	cout<<ans;
	return 0;
}
