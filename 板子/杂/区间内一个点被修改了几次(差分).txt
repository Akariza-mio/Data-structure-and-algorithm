#include<bits/stdc++.h>
using namespace std;
#define AKARIZA ios::sync_with_stdio(false);cin.tie(0);
using ll=long long;
using db=double;
const ll maxn=5e5+10,mod=998244353;
int dif[maxn];
int main(){
	AKARIZA
	int n,m;
	cin>>n>>m;
	string s,t;
	cin>>s>>t;
	for(int i=0;i<m;++i){
		int l,r;
		cin>>l>>r;
		dif[--l]+=1;
		dif[r]-=1;
	}
	string ans="";
	int tp=0;
	for(int i=0;i<n;i++){
		tp+=dif[i];
		if(tp%2==0)ans+=s[i];
		else ans+=t[i];
	}
	cout<<ans<<endl;
	return 0;
}
