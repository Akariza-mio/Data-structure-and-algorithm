#include<bits/stdc++.h>
using namespace std;
using ll=long long;
int q1(){
	cout<<1<<endl;
	cout.flush();
	int ans;
	cin>>ans;
	if(ans==-2||ans==0)exit(0);
	return ans;
}
int q2(int y){
	cout<<y<<endl;
	cout.flush();
	int ans;
	cin>>ans;
	if(ans==-2||ans==0)exit(0);
	return ans;
}
int main(){
    int m,n;
	cin>>m>>n;
	vector<int> a(n);
	for(int i=0;i<n;i++){
		int tp=q1();
		if(tp==-1)a[i]=0;
	    else if(tp==1)a[i]=1;
		else return 0;
	}
	int l=1,r=m,cnt=0;
	while(l<=r){
		int mid=(l+r)/2;
		int y=q2(mid);
		if(y==1){
			if(a[cnt]==1){
				l=mid+1;
				cnt++;
				if(cnt>=n)cnt=0;
			}else if(a[cnt]==0){
				r=mid+1;
				cnt++;
				if(cnt>=n)cnt=0;
			}
		}else if(y==-1){
			if(a[cnt]==1){
				r=mid+1;
				cnt++;
				if(cnt>=n)cnt=0;
			}else if(a[cnt]==0){
				l=mid+1;
				cnt++;
				if(cnt>=n)cnt=0;
			}
		}else exit(0);
	}
	return 0;
}
