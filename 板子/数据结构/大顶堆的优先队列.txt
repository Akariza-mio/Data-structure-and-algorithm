#include<iostream>
#include<vector>
#include<set>
#include<unordered_map>
#include<queue>
using namespace std;
int n,k,p,cnt;
unordered_map<int,queue<int>> oc;
set<int> slot;
unordered_map<int,int> valid;
priority_queue<pair<int,int>> a;
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	cin>>n>>k>>p;
	vector<int> next(p,p+1);
	vector<int> P(p);
	for(int i=0;i<p;i++)cin>>P[i];
	for(int i=0;i<p;i++)oc[P[i]].push(i);
	for(int i=0;i<p;i++){
		oc[P[i]].pop();
		if(!oc[P[i]].empty()){
			next[i]=oc[P[i]].front();
		}
	}
	cnt=0;
	for(int i=0;i<p;i++){
		int card=P[i];
		if(slot.count(card)){
			valid[card]=next[i];
			a.push({next[i],card});
			continue;
		}
		cnt++;
		if(slot.size()==k){
			while(!a.empty()){
				auto[b,c]=a.top();
				a.pop();
				if(slot.count(c)&&valid[c]==b){
					slot.erase(c);
					valid.erase(c);
					break;
				}
			}
		}
		slot.insert(card);
		valid[card]=next[i];
		a.push({next[i],card});
	}
	cout<<cnt<<endl;
	return 0;
}
