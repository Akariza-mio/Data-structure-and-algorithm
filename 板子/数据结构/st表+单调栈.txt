#include<bits/stdc++.h>
using namespace std;
using ll=long long;
struct st_table{
	int n,logn;
	vector<vector<ll>> st;
	vector<int> log2;
	bool ismin;
	st_table(const vector<ll>& a,bool _ismin):n((int)a.size()),ismin(_ismin){
		logn=0;
		while((1<<logn)<=n)logn++;
		st.assign(n,vector<ll>(logn));
		log2.resize(n+1);
		log2[1]=0;
		for(int i=2;i<=n;i++)log2[i]=log2[i/2]+1;
		for(int i=0;i<n;i++)st[i][0]=a[i];
		for(int j=1;j<logn;j++) {
			for(int i=0;i+(1<<j)<=n;i++){
				if(ismin)st[i][j]=min(st[i][j-1],st[i+(1<<(j-1))][j-1]);
				else st[i][j]=max(st[i][j-1],st[i+(1<<(j-1))][j-1]);
			}
		}
	}
	ll ask(int l,int r){
		int len=r-l+1;
		int k=log2[len];
		if(ismin)return min(st[l][k],st[r-(1<<k)+1][k]);
		else return max(st[l][k],st[r-(1<<k)+1][k]);
	}
};
pair<vector<int>,vector<int>> range(const vector<ll>& a){
	int n=(int)a.size();
	vector<int> left(n),right(n);
	stack<int> s;
	for(int i=0;i<n;i++){
		while(!s.empty()&&a[s.top()]<=a[i])s.pop();
		left[i]=s.empty()?0:s.top()+1;
		s.push(i);
	}
	while(!s.empty())s.pop();
	for(int i=n-1;i>=0;i--){
		while(!s.empty()&&a[s.top()]<a[i])s.pop();
		right[i]=s.empty()?n:s.top();
		s.push(i);
	}
	return {left,right};
}
int main(){
	ios::sync_with_stdio(false);
	cin.tie(0);
	int T;
	cin>>T;
	while(T--){
		int n=0;
		cin>>n;
		int flag=1;
		vector<ll> a(n,0);
		for(int i=0;i<n;i++)cin>>a[i];
		vector<ll> sum(n+1,0);
		for(int i=1;i<=n;i++)sum[i]=sum[i-1]+a[i - 1];
		auto[left,right]=range(a);
		st_table maxx(sum,false);
		st_table minn(sum,true);
		for(int i=0;i<n;i++){
			int le=left[i],ri=right[i];
			if(le<=i&&i+1<=ri){
				ll minPre=minn.ask(le,i);
				ll maxPre=maxx.ask(i+1,ri);
				if(maxPre-minPre>a[i]){
				    flag=0;
				}
			}
		}
		if(flag)cout<<"YES"<<endl;
		else cout<<"NO"<<endl;
	}
	return 0;
}
