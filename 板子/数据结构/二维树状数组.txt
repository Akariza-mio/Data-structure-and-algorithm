#include<iostream>
using namespace std;
#define maxn 2053
#define ll long long
int n,m;
ll a[maxn][maxn],b[maxn][maxn],c[maxn][maxn],d[maxn][maxn];
int lb(int x){
	return x&(-x);
}
void mdf(ll x,ll y,ll k){
	for(int i=x;i<=n;i+=lb(i)){
		for(int j=y;j<=m;j+=lb(j)){
			a[i][j]+=k;
			b[i][j]+=x*k;
			c[i][j]+=y*k;
			d[i][j]+=x*y*k;
		}
	}
}
void range_mdf(ll x1,ll y1,ll x2,ll y2,ll k){
	mdf(x1,y1,k);
	mdf(x1,y2+1,-k);
	mdf(x2+1,y1,-k);
	mdf(x2+1,y2+1,k);
}
ll ask(ll x,ll y){
	ll ans=0;
	for(int i=x;i>0;i-=lb(i)){
		for(int j=y;j>0;j-=lb(j)){
			ans+=(x+1)*(y+1)*a[i][j]-(y+1)*b[i][j]-(x+1)*c[i][j]+d[i][j];
		}
	}
	return ans;
}
ll range_ask(ll x1,ll y1,ll x2,ll y2){
	return ask(x2,y2)+ask(x1-1,y1-1)-ask(x2,y1-1)-ask(x1-1,y2);
}
int main(){
	cin>>n>>m;
	int c;
	while(cin>>c){
		ll x1=0,y1=0,x2=0,y2=0,k=0;
		if(c==1){
			cin>>x1>>y1>>x2>>y2>>k;
			range_mdf(x1,y1,x2,y2,k);
		}
		else{
			cin>>x1>>y1>>x2>>y2;
			cout<<range_ask(x1,y1,x2,y2)<<endl;
		}
	}
	return 0;
}
